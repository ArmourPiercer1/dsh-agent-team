    '{"generation":1,"createdAt":"2026-08-29T12:00:00.000Z","blueprint":{"revision":"17","contentHash":"sha256:abc123","blueprintId":"AIUED-ALGO"},"rootSessionId":"session-1","schemaVersion":1}'
