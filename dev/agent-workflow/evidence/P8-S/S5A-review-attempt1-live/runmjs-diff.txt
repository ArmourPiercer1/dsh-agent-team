+/**
+ * A YAML scalar for the patch-file emitter (the DSH profile-patch dialect
+ * is js-yaml JSON_SCHEMA + !!js: every scalar here is plain JSON — strings
+ * JSON-quoted (the \n escapes make the blueprint a single-line quoted
+ * scalar), numbers/booleans/null bare).
+ * @param {unknown} v
+ * @returns {string}
+ */
+function yamlScalar(v) {
+  if (v === null) return 'null'
+  if (typeof v === 'string') return JSON.stringify(v)
+  return String(v)
+}
+
+/**
+ * The nested entries of an already-emitted `key:` line, at the given indent
+ * (2-space steps; the mountRows dialect).
+ * @param {unknown} value - a non-empty plain object or array.
+ * @param {number} indent
+ * @returns {Array<string>}
+ */
+function yamlValueLines(value, indent) {
+  if (Array.isArray(value)) {
+    return value.flatMap((item) => yamlEmitItem(item, indent))
+  }
+  return Object.entries(value).flatMap(([k, v]) => yamlEmit(k, v, indent))
+}
+
+/**
+ * Emit one `key: ...` pair: a scalar on the line, a block map/sequence for
+ * a non-empty compound (an empty compound stays inline as {} / []).
+ * @param {string} key
+ * @param {unknown} value
+ * @param {number} indent
+ * @returns {Array<string>}
+ */
+function yamlEmit(key, value, indent) {
+  const pad = '  '.repeat(indent)
+  if (value !== null && typeof value === 'object') {
+    const empty = Array.isArray(value) ? value.length === 0 : Object.keys(value).length === 0
+    if (empty) return [`${pad}${key}: ${Array.isArray(value) ? '[]' : '{}'}`]
+    return [`${pad}${key}:`, ...yamlValueLines(value, indent + 1)]
+  }
+  return [`${pad}${key}: ${yamlScalar(value)}`]
+}
+
+/**
+ * Emit one sequence item: `- scalar` on its own line, or `- firstKey: ...`
+ * with the rest of the object's keys one step deeper.
+ * @param {unknown} item
+ * @param {number} indent - the dash column (2-space steps).
+ * @returns {Array<string>}
+ */
+function yamlEmitItem(item, indent) {
+  const pad = '  '.repeat(indent)
+  if (item === null || typeof item !== 'object') {
+    return [`${pad}- ${yamlScalar(item)}`]
+  }
+  if (Array.isArray(item)) {
+    if (item.length === 0) return [`${pad}- []`]
+    return [`${pad}-`, ...yamlValueLines(item, indent + 1)]
+  }
+  const entries = Object.entries(item)
+  if (entries.length === 0) return [`${pad}- {}`]
+  const [firstKey, firstValue] = entries[0]
+  const firstCompound = firstValue !== null && typeof firstValue === 'object'
+  const firstEmpty = firstCompound && (Array.isArray(firstValue) ? firstValue.length === 0 : Object.keys(firstValue).length === 0)
+  const firstLines = firstCompound
+    ? firstEmpty
+      ? [`${pad}- ${firstKey}: ${Array.isArray(firstValue) ? '[]' : '{}'}`]
+      : [`${pad}- ${firstKey}:`, ...yamlValueLines(firstValue, indent + 2)]
+    : [`${pad}- ${firstKey}: ${yamlScalar(firstValue)}`]
+  return [...firstLines, ...entries.slice(1).flatMap(([k, v]) => yamlEmit(k, v, indent + 1))]
+}
